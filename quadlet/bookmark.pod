# =============================================================================
# Bookmark Manager Production Pod — Quadlet unit file
# Installs to: ~/.config/containers/systemd/bookmark.pod
# Contains: MariaDB + phpMyAdmin + API
# Port bindings:
#   11650 → API        (127.0.0.1 only — access via the local machine or SSH tunnel)
#   11651 → phpMyAdmin (127.0.0.1 only — access via SSH tunnel or reverse proxy)
# MariaDB is pod-internal only (no host port binding).
# =============================================================================

[Unit]
Description=Bookmark Manager production pod
After=default.target
After=network-online.target
Wants=network-online.target
# Allow up to 10 restart attempts over 5 minutes.
# Belt-and-suspenders: the real fix is the system-level drop-in in
# quadlet/system-dropin/after-network.conf which delays the user session
# until network-online.target is reached (see that file for install steps).
StartLimitIntervalSec=300
StartLimitBurst=10

[Pod]
# API — localhost only (127.0.0.1:11650 → container 11650)
PublishPort=127.0.0.1:11650:11650
# phpMyAdmin — localhost only
PublishPort=127.0.0.1:11651:80

[Install]
WantedBy=default.target
